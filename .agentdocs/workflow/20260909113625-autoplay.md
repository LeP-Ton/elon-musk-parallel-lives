## 实现与边界

### 自动选择

- 只考虑 requirements 满足、至少有一个合法正权重结果的选项。
- 人格契合度、当前压力、精力与自主冲突影响权重；种子生成稳定的关注重点，不修改角色基础人格。
- 使用独立 `autoplay` 随机用途，以策略版本、事件和历史长度为键；不预演结果，不挪用 outcome 随机流。
- 同种子、初始状态、选择策略、内容与人物版本可复现。手动接管会改变后续人生，内容或策略升级也可能改变重播路线。
- 无合法选项时报错并暂停；结局态保持原对象，不跳过场景、不自动重开。

### 播放与中断

- 选择页和结果页各使用一个时钟；阅读时长按文字量计算，支持 1×、2×、4×，每页至少两秒。
- 倒计时、预告选择及结果页的“这一刻的选择”让观众知道发生了什么。
- 暂停同步取消句柄并使回调失效；计时回调校验当前游戏、内容包、面板、错误与页面可见性快照。
- 手动选择、翻页、WebMCP 操作、面板、读档、导入、内容包切换、存档错误和后台事件都会阻止自动推进。
- React 已提交快照通过 layout effect 同步，避免并发渲染期间将未提交人生交给时钟。快捷键通过 effect event 读取最新状态。
- 从后台返回、关闭面板或读取存档后不自动恢复，需要观众重新播放。
- 计时器 effect 需要同步初始化倒计时与停止态，对其局部保留有说明的 React Compiler effect-set-state 规则例外；没有全局降低 lint 标准。

### 随机新人生与存档

- 第一次直接自动播放采用新的系统熵种子；暂停和倍速不重新生成种子。
- “随机开播”只打开带新种子的确认面板；确认后才替换自动存档，手动存档保留。
- 旧存档 `schemaVersion: 1` 不变，所有推进仍经过自动存档；不保存 playing、speed 或计时器状态。
- 故事到达现有五种阶段结局之一后停留。没有新增剧情、后端或模型 API。

## 测试用例与结果

### TC-001 规则与多结局回归

- 类型：自动化功能测试；优先级：高；模块：自动选择策略、运行时。
- 步骤：对 `AUTO-0` 至 `AUTO-999` 运行完整自动人生，每一步与原手动入口的输出严格比较，检查选项条件与终局条件。
- 预期：全部在限制步数内结束；至少覆盖当前全部结局；不因自动模式绕过原规则。
- 结果：通过。1,000 轮全部结束，最长 18 个事件。
- 分布：独立路线 independent 343、创业 founder 191、研究 research 221、游戏 games 194、工程师 engineer 51。
- 该分布是固定样本回归，不是现实概率或面向观众的结局配额。

### TC-002 可复现与随机隔离

- 类型：自动化单元测试；优先级：高。
- 步骤：重复同种子自动人生；反复预告选择、计算不同倍速阅读时长、执行 cosmetic 抽样。
- 预期：相同状态产生相同选择和结果，不修改原始输入。
- 结果：通过。

### TC-003 约束与错误停止

- 类型：自动化单元测试；优先级：高。
- 步骤：加入锁定选项和结果全为零的选项；清空合法选项；设置高压力、低精力、自主冲突状态。
- 预期：过滤不可执行选择，无路可走时报错；当前状态影响选择权重。
- 结果：通过。

### TC-004 旧存档、接管与内容扩展

- 类型：自动化契约测试；优先级：高。
- 步骤：在选择页和结果页序列化并读取；自动后手动选择再自动续播；注册独立测试包。
- 预期：存档格式兼容、续播可复现、手动历史保留，新内容包无需修改自动策略即可执行。
- 结果：通过。此项测试引擎/存档层，不冒充真实 UI 读档点击测试。

### TC-005 计时器生命周期

- 步骤：模拟到期、取消、失效快照、倒计时回调期间接管、规则异常及替换旧时钟。
- 预期：一次只执行一步；取消或失效后无推进；错误只上报一次；旧时钟不泄漏。
- 结果：通过。测试中的暂停/接管/读档/切包/卸载等名称对应取消契约，后台/面板/错误等名称对应快照契约，并非实际浏览器事件联调。

+## 自动播放：观众模式
+
+点击故事上方的「自动播放」，人物会在合法选项中，根据人格倾向、压力、精力、自主冲突及种子机遇作出选择，并依次展示结果、翻页到下一场景。它使用原有的概率与结局规则，不提前挑选结局。首次直接开播会生成随机种子，之后可续播当前人生。

+import type { Choice, Game, Registry } from '../game/types';
+import { matches } from './conditions';
+import { calculateChoiceFit } from './personality';
+import { outcomeWeights } from './probability';
+import { random, weightedIndex } from './rng';
+import { advance, choose, narrativeText } from './runtime';
+
+export const AUTOPLAY_POLICY = 'autoplay-v1';
+export type PlaybackSpeed = 1 | 2 | 4;
+
+/** 独立选择策略：不预演结果、不挑选结局，也不消耗结果判定的随机流。 */
+export function automaticChoices(game: Game, registry: Registry) {
+  if (game.phase !== 'choice') return [];
+  const event = registry.events.find((e) => e.id === game.currentEventId);
+  if (!event) throw new Error('自动播放找不到当前场景。');
+  const state = game.state;
+  return (
+    event.choices
+      .filter((choice) => matches(choice.requirements, state))
+      // 条件可能让某个选项的所有结果都失效，此时不将它加入自动候选。
+      .filter((choice) =>
+        outcomeWeights(choice, state).some((weight) => weight > 0),
+      )
+      .map((choice) => {
+        const fit = calculateChoiceFit(choice, state).score;
+        const affinity = Object.entries(choice.affinity ?? {});
+        const magnitude = affinity.reduce(
+          (sum, [, value]) => sum + Math.abs(value),
+          0,
+        );
+        // 每轮的关注重点由种子稳定生成；不修改角色本身的人格参数。
+        const emphasis = magnitude
+          ? affinity.reduce(
+              (sum, [axis, value]) =>
+                sum +
+                value *
+                  (random(
+                    state.seed,
+                    'autoplay',
+                    `${AUTOPLAY_POLICY}:focus:${axis}`,
+                  ) *
+                    2 -
+                    1),
+              0,
+            ) / magnitude
+          : 0;
+        const caution = (state.stress + 100 - state.energy) / 200;
+        const need = choice.affinity ?? {};
+        const context =
+          caution *
+            ((need.stabilityNeed ?? 0) - (need.riskTolerance ?? 0)) *
+            0.6 +
+          (state.autonomyConflict / 100) * (need.autonomy ?? 0) * 0.8;
+        const weight = Math.exp(
+          Math.max(
+            -3,
+            Math.min(3, (fit - 50) / 80 + emphasis * 0.75 + context),
+          ),
+        );
+        return { choice, weight, fit };
+      })
+  );
+}

+export function selectAutomaticChoice(
+  game: Game,
+  registry: Registry,
+): Choice | null {
+  if (game.phase !== 'choice') return null;
+  const candidates = automaticChoices(game, registry);
+  if (!candidates.length)
+    throw new Error('当前没有可执行的自动选择，已暂停。请检查内容或读取存档。');
+  const roll = random(
+    game.state.seed,
+    'autoplay',
+    `${AUTOPLAY_POLICY}:choice:${game.currentEventId}`,
+    game.state.history.length,
+  );
+  return candidates[
+    weightedIndex(
+      candidates.map((row) => row.weight),
+      roll,
+    )
+  ].choice;
+}

+export function stepAutomatically(game: Game, registry: Registry): Game {
+  if (game.phase === 'ending') return game;
+  if (game.phase === 'result') return advance(game, registry);
+  return choose(game, registry, selectAutomaticChoice(game, registry)!.id);

+  registry: Registry,
+  speed: PlaybackSpeed,

+  const event = registry.events.find((e) => e.id === game.currentEventId);

+type ClockCallbacks = {
+  isCurrent: () => boolean;
+  onRemaining: (milliseconds: number) => void;
+  onElapsed: () => void;
+  onError: (error: unknown) => void;
+};
+
+/** 一次性页面时钟；暂停、接管和卸载都取消同一个句柄，过期回调不能推进。 */
+export function startPlaybackClock(delay: number, callbacks: ClockCallbacks) {
+  if (!Number.isFinite(delay) || delay <= 0)
+    throw new Error('播放间隔必须大于零。');
+  const deadline = performance.now() + delay;
+  let cancelled = false;
+  const cancel = () => {
+    cancelled = true;
+    clearInterval(timer);
+  };
+  const timer = setInterval(() => {
+    if (cancelled) return;
+    try {
+      if (!callbacks.isCurrent()) return cancel();
+      const remaining = Math.max(0, deadline - performance.now());
+      callbacks.onRemaining(remaining);
+      if (remaining === 0 && !cancelled && callbacks.isCurrent()) {
+        cancel();
+        callbacks.onElapsed();
+      }
+    } catch (error) {
+      cancel();
+      callbacks.onError(error);
+    }
+  }, 200);
+  return cancel;
+}

+  useLayoutEffect(() => {
+    current.current = input;
+  });
+  const { game, registry, blocked } = input;
+  const plan = useMemo(() => {
+    try {
+      return { choice: selectAutomaticChoice(game, registry), error: null };
+    } catch (error) {
+      return { choice: null, error };
+    }
+  }, [game, registry]);
+
+  function pause() {
+    // 同步失效，不能等 React 下一次渲染后才取消已排队的自动动作。
+    playingRef.current = false;
+    cancelRef.current();
+    setPlaying(false);
+    setRemaining(0);
+  }
+  function start() {
+    // 开新人生和关闭面板可能与启动同批提交，合法性由下一次 effect 的新快照检查。
+    playingRef.current = true;
+    setPlaying(true);
+  }
+  function changeSpeed(next: PlaybackSpeed) {
+    if (![1, 2, 4].includes(next) || next === speed) return;
+    cancelRef.current();
+    setSpeed(next);
+  }
+  useEffect(() => {
+    const onVisibility = () => {
+      if (document.hidden) pause();
+    };
+    document.addEventListener('visibilitychange', onVisibility);
+    return () => {
+      document.removeEventListener('visibilitychange', onVisibility);
+      playingRef.current = false;
+      cancelRef.current();
+    };
+  }, []);
+  // 此 effect 管理外部时钟，同时初始化倒计时和同步停止态，不能等待首次 tick 才阻止过期动作。
+  /* eslint-disable react/react-compiler */
+  useEffect(() => {
+    if (!playing) return;
+    if (blocked || game.phase === 'ending' || document.hidden) {
+      pause();
+      return;
+    }
+    try {
+      if (plan.error) throw plan.error;
+      const duration = readingDelay(game, registry, speed);
+      setRemaining(duration);
+      cancelRef.current = startPlaybackClock(duration, {
+        isCurrent: () =>
+          playingRef.current &&
+          !document.hidden &&
+          !current.current.blocked &&
+          current.current.game === game &&
+          current.current.registry === registry,
+        onRemaining: setRemaining,
+        onElapsed: () =>
+          current.current.onStep(stepAutomatically(game, registry)),
+        onError: (error) => {
+          pause();
+          current.current.onError(error);
+        },
+      });
+      return () => cancelRef.current();
+    } catch (error) {
+      pause();
+      current.current.onError(error);
+    }
+  }, [game, registry, blocked, playing, speed, plan]);
+  /* eslint-enable react/react-compiler */
+  return {
+    playing,
+    speed,
+    remaining,
+    plannedChoice: playing ? plan.choice : null,
+    start,
+    pause,
+    changeSpeed,
+  };
+}
