## 关键风险
- 私有源码仓库启用 Pages 需要 GitHub 支持的套餐；账号查询没有返回套餐信息，不能推断已有 Pro。
- Pages 页面与源码仓库的可见性是两个不同选择；Pages 页面将对外公开。
- Vinext 的子目录静态导出带有基础路径，上传 Pages 时必须去掉这一层目录。
- Sites 与 Pages 存档因域名隔离，需要用户导出后导入；没有云端同步。
- 构建预渲染临时监听 localhost，需要允许该本机端口；首次沙箱内构建被 EPERM 阻止，经工具审批后通过。

+/** 为公开资源补上部署路径；内容包始终保留与平台无关的资源地址。 */
+export function assetUrl(
+  path: string | undefined,
+  basePath = process.env.NEXT_PUBLIC_BASE_PATH ?? '',
+): string | undefined {
+  if (!path || !path.startsWith('/') || path.startsWith('//')) return path;
+  return `${basePath.replace(/\/$/, '')}${path}`;
+}
